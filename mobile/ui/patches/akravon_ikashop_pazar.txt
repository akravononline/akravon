AKR_COLS = 6
AKR_ROWS = 6
AKR_MAX_ITEMS = 20
AKR_CELL = 56


def AkrCellToSlot(cell):
	row = cell // ikashop.SHOP_GRID_WIDTH
	col = cell % ikashop.SHOP_GRID_WIDTH
	if row < AKR_ROWS and col < AKR_COLS:
		return row * AKR_COLS + col
	return None


def AkrSlotToCell(slot):
	return (slot // AKR_COLS) * ikashop.SHOP_GRID_WIDTH + slot % AKR_COLS


def AkrConvertItems(items):
	# sunucu hucresi (10 sutun) -> pencere yuvasi (6 sutun); disarida kalan eski kayitlar gizlenir
	result = {}
	for it in items:
		slot = AkrCellToSlot(it['cell'])
		if slot is None:
			continue
		it['cell'] = slot
		result[slot] = it
	return result
